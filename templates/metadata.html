databricks.yml

bundle:
  name: customer-analytics

include:
  - resources/*.yml

variables:
  cost_center:
    default: "CC-4100"
  project_id:
    default: "PRJ-001"
  change_id:
    default: "CHG-2026-0042"
  release_version:
    default: "v1.0.184"

targets:
  dev:
    mode: development
  qa:
    mode: production
  prod:
    mode: production

resources:
  jobs:
    customer_analytics_job:
      name: customer-analytics-${var.project_id}
      tasks:
        - task_key: customer_processing
          notebook_task:
            notebook_path: /Workspace/customer-analytics
